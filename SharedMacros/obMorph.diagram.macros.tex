% nodenames are constructed and have an optional instance
%  parameter so that two copies of the same content 
%  can be separately identified.

% Construct node names
\newcommand{\ob}[3]{#1#2#3}
\newcommand{\bob}[1]{#1}
\newcommand{\obp}[3]{#1#2#3p}
\newcommand{\obs}[3]{#1#2#3s}
\newcommand{\obpp}[3]{#1#2#3pp}
\newcommand{\obss}[3]{#1#2#3ss}

% Canonical node-name constructor:
% {instance}{object}{subscript}{operator}
\newcommand{\nodename}[4]{#1#2#3#4}

% Object
\NewDocumentCommand{\Ob}{O{} m m}{%
  \Rnode{\nodename{#1}{#2}{#3}{}}{#2_{#3}}%
}

% Bare object, without subscript
\NewDocumentCommand{\bOb}{O{} m}{%
  \Rnode{\nodename{#1}{#2}{}{}}{#2}%
}

% Predecessor
\NewDocumentCommand{\Obp}{O{} m m}{%
  \Rnode{\nodename{#1}{#2}{#3}{p}}{#2_{#3-1}}%
}

% Successor
\NewDocumentCommand{\Obs}{O{} m m}{%
  \Rnode{\nodename{#1}{#2}{#3}{s}}{#2_{#3+1}}%
}

% Two predecessors
\NewDocumentCommand{\Obpp}{O{} m m}{%
  \Rnode{\nodename{#1}{#2}{#3}{pp}}{#2_{#3-2}}%
}

% Two successors
\NewDocumentCommand{\Obss}{O{} m m}{%
  \Rnode{\nodename{#1}{#2}{#3}{ss}}{#2_{#3+2}}%
}

\NewDocumentCommand{\zigBulletObject}{m m m}
  {\Rnode{L\ob{#1}{#2}{#3}}{\strut} %left connection point

  }

\ExplSyntaxOn
\NewDocumentCommand{\zigLabel}{m m m}
 {
  \prop_get:NeNTF \l_zig_label_position_prop {#1} \l_tmpa_tl
   { }
   { \tl_set:Nn \l_tmpa_tl {#2} }

  \tl_if_eq:VnTF \l_tmpa_tl {a}
   { \alabel{#3}[0.5][0.4] }
   {
    \tl_if_eq:VnTF \l_tmpa_tl {b}
     { \blabel{#3}[0.5][0.4] }
     {
      \PackageError{zigzag}
       {Unknown label side}
       {Use a or b.}
     }
   }
 }

% \zigMorphism{source}{target}{instance}{name}{index}
% Forward morphism: f_i

 \NewDocumentCommand{\zigMorphism}{m m m m m}
 {
  \ncarr{R#1}{#2}
  \zigLabel
   {#3#4#5}
   {a}
   {
    \MorphismContent{#3#4#5}
     {#4_{\IndexTranslation{#5}{#5}}}
   }
 }

% \zigReverseMorphism{source}{target}{instance}{name}{index}
% Reverse morphism: f'_i

\NewDocumentCommand{\zigReverseMorphism}{m m m m m}
 {
  \ncarr{L#1}{#2}
  \zigLabel
   {#3#4prime#5}
   {b}
   {
    \MorphismContent{#3#4prime#5}
     {#4'_{\IndexTranslation{#5}{#5}}}
   }
 }

% zigzagOneObjects[instance]{x}{y}
\NewDocumentCommand{\zigzagOneObjects}{O{} m m}{
  \begin{array}{c c c c}
   & \zigObject{#1}{#3}{1}
   \\[0.8cm]

  \end{array}
}

   & \zigBulletObject{#1}{#3}{1}
   \\[0.8cm]
   \zigBulletObject{#1}{#2}{1} &
  \end{array}
}

% zigzagOneMorphisms[instance]{x}{y}{f}
\NewDocumentCommand{\zigzagOneMorphisms}{O{} m m m}{
  \begin{arrows}
    \zigMorphism{\ob{#1}{#2}{1}}{\ob{#1}{#3}{1}}
              {#1}{#4}{1}
  \end{arrows}
}

% \zigzagTwo[instance]{x}{y}{f}
% draws a zigzag of length two <f1,f'1,f2> 
% where f1 : x1 -> y1
%       f'1: x2 -> y1
%       f2 : x2 -> y2
%
% overide xi  using \SetObjectContent{\ob{instance}{x}{i}}{...}
% override fi using \SetMorphismContent{fi}{...}
%                or \SetMorphismContent{fprimei}{...} 

\NewDocumentCommand{\zigzagTwoObjects}{O{} m m}{
  \begin{array}{c c c c}
   & \zigObject{#1}{#3}{1}
   &&
     \zigObject{#1}{#3}{2}
   \\[0.8cm]
   \zigObject{#1}{#2}{1}
   &&
   \zigObject{#1}{#2}{2}
  \end{array}
}

\NewDocumentCommand{\zigzagTwoObjectsBulleted}{O{} m m}{
  \begin{array}{c c c c}
   & \zigBulletObject{#1}{#3}{1}
   &&
     \zigBulletObject{#1}{#3}{2}
   \\[0.8cm]
   \zigBulletObject{#1}{#2}{1}
   &&
   \zigBulletObject{#1}{#2}{2}
  \end{array}
}

\NewDocumentCommand{\zigzagTwoMorphisms}{O{} m m m}{
  \begin{arrows}
    \zigzagOneMorphisms[#1]{#2}{#3}{#4}
    \zigReverseMorphism{\ob{#1}{#2}{2}}{\ob{#1}{#3}{1}}
              {#1}{#4}{1}
    \zigMorphism{\ob{#1}{#2}{2}}{\ob{#1}{#3}{2}}
              {#1}{#4}{2}
  \end{arrows}
}

\NewDocumentCommand{\zigzagTwo}{O{} m m m}{
  \zigzagTwoObjects[#1]{#2}{#3}
  \zigzagTwoMorphisms[#1]{#2}{#3}{#4}
}

\NewDocumentCommand{\zigzagTwoBulleted}{O{} m m m}{
  \zigzagTwoObjectsBulleted[#1]{#2}{#3}
  \zigzagTwoMorphisms[#1]{#2}{#3}{#4}
}
