% zig one is  f1 : x1 -> y1 
% zig one is  f'1 : x2 -> y1
\NewDocumentCommand{\zigzagzigTwo}{m m m o}
{
\begin{array} {c c c c c}
 &\Ob{#2}{1}&&\Ob{#2}{2}          &\\[0.8cm]
\IfNoValueTF{#4}{\Ob{#1}{1}}{\Rnode{#11}{}}&&\Ob{#1}{2}&&\Ob{#1}{3} 
\end{array}
\begin{arrows}
\ncarr{#11}{#21} \alabel{ 
\IfNoValueTF{#4}
    {}
    {#4 \circ} #3_1
}[0.45][0.4]
\ncarr{#12}{#21} \blabel{#3'_1}[0.45][0.4]
\ncarr{#12}{#22} \alabel{#3_2}[0.45][0.4]
\ncarr{#13}{#22} \blabel{#3'_2}[0.45][0.4]
\end{arrows}
}

    {}
    {\circ #4}
}[0.45][0.4]
\ncarr{#12}{#22} \alabel{#3_2}[0.45][0.4]
\ncarr{#12}{#21} \blabel{#3'_1}[0.45][0.4]
\end{arrows}
}
