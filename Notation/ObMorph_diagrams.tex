\subsection{Aggregated}

\begin{showexample}
\begin{wrappedZigzag}{x}{y}{f}
    \SetIndexTranslation{1}{i-1}
    \SetIndexTranslation{2}{i}
    \SetIndexTranslation{3}{i+1}
    \arraycolsep=12pt
    \zigzagThree{x}{y}{f}
\end{wrappedZigzag}
\end{showexample}


\begin{showexample}
\begin{wrappedZigzag}{x}{y}{f}
    \SetIndexTranslation{1}{i-1}
    \SetIndexTranslation{2}{i+1}
    \SetObjectContent{x2}{\kern0.25cm x_{i+1}\kern0.15cm} 
    \SetMorphismContent{fprime1}{f'_{i-1}\circ f_i \circ f'_i}
    \SetMorphismLabelPosition{f2}{b} %flips the label out of the way 
                                     % of the lengthy fprime2 label
    \arraycolsep=10pt
    \zigzagTwo{x}{y}{f}
\end{wrappedZigzag}
\end{showexample}
