\section{Bisupport Categories}

\begin{definition}
A \term{bisupport category} is a category $\catc$ equipped, for each
morphism $f:a\morph b$, with morphisms
\[
\rst{f}:a\morph a
\qquad\text{and}\qquad
\rng{f}:b\morph b,
\]
satisfying the support and cosupport axioms and in addition the following compatibility  conditions for any morphism $f$

\noindent [SC (i)]
\[
\rst{\rng{f}}=\rng{f},
\]

\noindent [SC (ii)]
\[
\rng{\rst{f}}=\rst{f}.
\]

The duality is maintained by these two compatibility conditions because they are dual to one another.
\end{definition}

With bisupport defined in this way then the opposite category $\catc^{op}$
of a category \catcw is also a bisupport category if we switch the operators. Because of the duality, to each property of a bisupport category, there is a dual property which is established, free of charge, by the duality. 


\begin{lemma}
\llabel{bisupportLemma}
 If \catcw is a bisupport category then
 \begin{enumerate}[(i)]
 \item
 For $\sequentialdiag{a}{b}{c}{f}{g}$ in \catcw  then
\[
\rng{f \circ \rst{g}}
= \rng{f} \circ \rst{g}
= \rst{g} \circ \rng{f}
= \rst{\rng{f} \circ g}.
\]
\item
For \fgsourcediag in \catc, 
\[
\rst{f \circ \rng{\rst{g} \circ f}} \circ g = \rst{f} \circ g.
\]
\item
For \fgsinkdiag in \catc, 
\[
f \circ \rng{\rst{g \circ \rng{f}} \circ g} = f \circ \rng{g}.
\]
 \item %(iv)
If $\sequentialdiag{a}{b}{c}{f}{g}$ in \catcw  and
$\rng{f} = \rst{g}$ then 
\[
\rst{f \circ g} = \rst{f}
\]
and
\[
\rng{f \circ g}  = \rng{g}.
\]
\item %{v}
For $\triplesequentialdiag{a}{b}{a}{b}{f}{g}{h}$ in \catcw \newline
if $f$ and $g$ fit together snugly in the sense that 
\[
\rng{f} = \rst{g}
\]
and if $f \circ g$ is a support idempotent so that we have
\[
\rst{f \circ g} = f \circ g = \rng{f \circ g}
\]
and if $g$ and $h$ fit together snugly so that 
\[
\rng{g} = \rst{h}
\]
and if $g \circ h$ is a support idempotent so that we have
\[
\rst{g \circ h} = g \circ h = \rng{g \circ h}
\]
then
\[
f = h.
\]
\end{enumerate}
\end{lemma}
\begin{proof}\leavevmode\par
\begin{enumerate}[(i)]
\item %(i)
That $\rng{f \circ \rst{g}}
= \rng{f} \circ \rst{g}$ follows because
\begin{align*}
\rng{f \circ \rst{g}}
&= \rng{f \circ \rng{\rst{g}}}
&&\mbox{by SC(ii),}\\
&= \rng{f} \circ \rng{\rst{g}}
&&\mbox{by C.3,}\\
&=\rng{f} \circ \rst{g}
&&\mbox{by SC(ii).}
\end{align*}
That $\rst{\rng{f} \circ g}= \rst{g} \circ \rng{f}$
follows on by duality.

Finally, $\rng{f} \circ \rst{g} = \rst{g} \circ \rng{f}$ 
follows by using SC(i) and S.2.
\item %(ii) old{iv}
\begin{align*}
\rst{f \circ \rng{\rst{g} \circ f}} \circ g
               &= \rst{f \circ \rng{\rst{g} \circ f}} \circ \rst{g} \circ g
               && \mbox{by S.1,}\\
               &= \rst{\rst{f \circ \rng{\rst{g} \circ f}} \circ \rst{g}} \circ g
               && \mbox{by S.3 then S.4,}\\
               &= \rst{\rst{g} \circ \rst{f \circ \rng{\rst{g} \circ f}}} \circ g
               && \mbox{by S.2,}\\
               &= \rst{\rst{\rst{g} \circ f \circ \rng{\rst{g} \circ f}}} \circ g
               && \mbox{by S.3,}\\
               &= \rst{\rst{g} \circ f \circ \rng{\rst{g} \circ f}} \circ g
               && \mbox{by lemma \ref{supportLemma} (ii),}\\
               &= \rst{\rst{g} \circ f } \circ g
               && \mbox{by C.1,}\\
               &= \rst{g} \circ \rst{f}  \circ g
               && \mbox{by S.3,}\\
               &=  \rst{f}  \circ \rst{g}  \circ g
               && \mbox{by S.2,}\\
               &= \rst{f}  \circ g
               && \mbox{by S.1, as required.}
\end{align*}
\item %(iii)
This is the dual to (ii).
\item %{iv}
Assume that $\rng{f} = \rst{g}$ then
\begin{align*}
\rst{f \circ g} &= \rst{f \circ \rst{g}} 
                             && \mbox{by S.4,} \\
                &= \rst{f \circ \rng{f}}
                             && \mbox{by the initial assumption,} \\
                &= \rst{f}   && \mbox{by C.1.}
\end{align*} 
That $\rng{f \circ g} = \rng{g}$ follows by duality.
\item %(v)
Firstly,
\begin{align*}
f &= f \circ \rng{f} 
                && \mbox{by axiom C.1,} \\
  &= f \circ \rst{g}
                && \mbox{since $f$ is snug with $g$,} \\
  &= f \circ \rst{g \circ h}
                && \mbox{by (iv) since $g$ is snug with $h$,} \\
  &= f \circ g \circ h
                && \mbox{since $g \circ h$ is a support idempotent.} 
\end{align*}
Secondly, by duality, we have that $h = f \circ g \circ h$. 

Thus $f= f \circ g \circ h = h$.
\end{enumerate}
\end{proof}
