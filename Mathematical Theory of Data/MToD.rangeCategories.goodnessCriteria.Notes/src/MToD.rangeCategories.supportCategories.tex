\section{Support Categories}

\begin{definition}
A \term{support category} is a category $\catc$ equipped, for each
morphism $f:a\morph b$, with a morphism
\[
\rst{f}:a\morph a
\]
called  the \term{support} of $f$.


The support operator satisfies the following axioms.

\noindent [S.1]\quad
If $f : a \morph b$, then
\[
\rst{f}\circ f=f.
\]

\noindent [S.2]\quad
If \fgsourcediag in $\catc$, then
\[
\rst{g}\circ\rst{f} =\rst{f}\circ\rst{g}.
\]

\noindent [S.3]\quad
If \fgsourcediag in $\catc$, then
\[
\rst{\rst{f}\circ g} = \rst{f}\circ\rst{g}.
\]

\noindent [S.4]\quad
If $\sequentialdiag{a}{b}{c}{f}{g}$ in $\catc$, then
\[
\rst{f\circ g} = \rst{f\circ\rst{g}}.
\]
\end{definition}


\begin{lemma}
\llabel{supportLemma}
If \catcw is a support category then
\begin{enumerate}[(i)]
\item For any object $a$ of \catc, 
\[
\rst{id_a}= id_a,
\]
\item For any $f:a \morph b$ in \catc, 
\[
\rst{\rst{f}}=\rst{f},
\]
\item For any $f:a \morph b$ in \catc, 
\[
\rst{f} \circ \rst{f} = \rst{f},
\]
\item For $\sequentialdiag{a}{b}{c}{f}{g}$ in \catc,
 \commentary {i.e. $\rst {f \circ g} \sleq \rst{f}$.}
\[
\rst{f \circ g} \circ \rst{f} = \rst{f \circ g},
\]
\item %was newSupportLemma
For \fgsourcediag in \catc,
\[
\rst{\rst{g} \circ f} \circ g = \rst{f} \circ g,
\] 
\item
For  $\binarysinkPreceeded{a}{b}{c}{d}{f}{g}{h}$
in \catc,
\[
\rst{ f \circ \rst{g} \circ h } = \rst{ f \circ \rst{h} \circ g }.
\]
\end{enumerate}
\end{lemma}
\begin{proof}\leavevmode\par
\begin{enumerate}[(i)]
\item Follows from S.1.
\item Follows from S.4 with $id_B$ substituted for $g$.
\item Follows from S.1, with $\rst{f}$ substituted for $f$, and then (ii).
\item %(iv) scrap paper
\begin{align*}
\rst{f \circ g} \circ \rst{f}
&= \rst{f} \circ \rst{f \circ g}
&&\mbox{by S.2,}\\
&= \rst{ \rst{f} \circ f \circ g}
&&\mbox{by S.3,}\\
&= \rst{ f \circ g}
&&\mbox{by S.1.}\\
\end{align*}
\item %(v)
\begin{align*}
\rst{\rst{g} \circ f} \circ g
    &= \rst{g} \circ \rst{f} \circ g && \mbox{by S.3,}\\
    &= \rst{f} \circ \rst{g} \circ g  && \mbox{by S.2,}\\
    &= \rst{f} \circ g               && \mbox{by S.1.}
\end{align*}
\item %{vi)}
\begin{align*}
\rst{f \circ \rst{g} \circ h}
&= \rst{f \circ \rst{\rst{g} \circ h}}
&&\mbox{by S.4}\\
&= \rst{f \circ \rst{g} \circ \rst{h}}
&&\mbox{by S.3}\\
&= \rst{f \circ \rst{h} \circ \rst{g}}
&&\mbox{by S.2}\\
&= \rst{f \circ \rst{\rst{h} \circ g}}
&&\mbox{by S.3}\\
&=\rst{f \circ \rst{h} \circ g}
&&\mbox{by S.4}
\end{align*}
\end{enumerate}
\end{proof}

\subsection{Support Ordering of $Hom(A,B)$}

In a category with support, \catc, we can define a partial ordering on each hom set
$Hom_\catc(a,b)$ by defining :
$$f \sleq g \mbox{ iff } f = \rst{f} \circ g.$$

The relation $\sleq$, so defined, is antisymmetric because if
\begin{equation}
\label{fleqg}
f = \rst{f} \circ g
\end{equation}
and
\begin{equation}
\label{gleqf}
g = \rst{g} \circ f
\end{equation}
then
\begin{align*}
f &= \rst{f} \circ g &&\mbox {this is (\ref{fleqg}),}        \\
  &= \rst{g} \circ \rst{f} \circ g &&\mbox{by R.1 and R.2,}\\
  &= \rst{g} \circ f               &&\mbox{using (\ref{fleqg}) again,} \\
  &= g                             &&\mbox{(by (\ref{gleqf}).} 
\end{align*}

When $f$ and $g$ are partial functions and
considered as morphisms in the category $\Par$ of sets and partial functions,  
then  $f \leq g$ iff  whenever $f$ is defined at an $x$,  then $g$ too is defined at x and $f(x) = g(x)$.
