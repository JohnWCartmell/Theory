\section{Cosupport Categories}

\begin{definition}
A \term{cosupport category} is a category $\catc$ equipped, for each
morphism $f:a\morph b$, with morphisms
\[
\rng{f}:b\morph b,
\]
called the \term{cosupport} of $f$.

The cosupport operator are the duals to the support axioms. We denote by $\mathrm{C}.1$--$\mathrm{C}.4$.
Thus, explicitly,

\noindent [C.1]\quad
If $f : a \morph b$, then
\[
f\circ\rng{f} = f.
\]

\noindent [C.2]\quad
If \fgsinkdiag in $\catc$, then
\[
\rng{f}\circ\rng{g}
=
\rng{g}\circ\rng{f}.
\]

\noindent [C.3]\quad
If \fgsinkdiag in $\catc$, then
\[
\rng{\rng{g}\circ f} = \rng{g}\circ\rng{f}.
\]

\noindent [C.4]\quad
If $\sequentialdiag{a}{b}{c}{f}{g}$ in $\catc$, then
\[
\rng{f\circ g} = \rng{\rng{f}\circ g}.
\]
\end{definition}

\begin{lemma}
\llabel{copyOfnewSupportLemma}
If \fgsourcediag in a category with support \catcw then
\[
\rst{\rst{g} \circ f} \circ g = \rst{f} \circ g
\]
\end{lemma}
\begin{proof}
\begin{align*}
\rst{\rst{g} \circ f} \circ g
    &= \rst{g} \circ \rst{f} \circ g && \mbox{by R.3,}\\
    &= \rst{f} \circ \rst{g} \circ g  && \mbox{by R.2,}\\
    &= \rst{f} \circ g               && \mbox{by R.1.}
\end{align*}
\end{proof}

\begin{lemma}
\llabel{copyOfunirangelemma}
If $f:a \morph a$ in a range category \catcw then for any morphism 
$f: a \morph b$
\begin{enumerate}[(i)] 
\item $\rng{\rng{f}}=\rng{f}$,
\item $\rng{f} \comp \rng{f} = \rng{f}$,
\item \llabel{lem:unirange:hatbar} $\rng{\rst{f}}=\rst{f}$.
\end{enumerate}
\end{lemma}
\begin{proof}
Straightforward.
\end{proof}

and $h \leq f \circ g$
in \highlight{some kind of} category \catcw
then 
\begin{enumerate}[(i)]
\item
$h$ can be factored as
$h = h_L \circ h_R$ where 
$h_L \leq f$
and 
   $h_R \leq g$
by defining
\[
h_L = \rst{h} \circ f \circ \rst{g \circ \rng{h}}
\]
and  
\[
h_R = \rng{\rst{h} \circ f} \circ g \circ \rng{h}.
\] 
\item
If h is so factored then
\[
\rst{h_L}=\rst{h}
\]
and
\[
\rng{h_R}=\rng{h}.
\]
\end{enumerate}
\end{construction}

\begin{enumerate}[(i)]
\item %(i)
That $h_L \leq f$
and $h_R \leq g$ are the case
follows from the form of the definitions
given of $h_L$ and $h_R$ as restriction idempotents
composed with $f$, respectively, $g$.
(see observation of lemma ???
 which says $\rst{i} \circ f \leq f$ and $f \circ \rst{j} \leq f$).

That $h = h_L \circ h_R$ follows 
because restriction idempotents commute (R.2) and by R.1 and RR.2. 

\item %(ii)
Firstly, $\rst{h_L}=\rst{h}$ because
\begin{align*}
\rst{\rst{h} \circ f \circ \rst{g \circ \rng{h}}}
    &= \rst{\rst{h} \circ f \circ g \circ \rng{h}}
        && \mbox{by wR.4,}\\
    &= \rst{h \circ \rng{h}}
        && \mbox{because $h \leq f \circ g$,}\\
    &= \rst{h} && \mbox{by RR.2.}
\end{align*}

\noindent 
Next, $\rng{h_R}=\rng{h}$ because
\begin{align*}
\rng{\rng{\rst{h} \circ f} \circ g \circ \rng{h}}
    &= \rng{\rst{h} \circ f \circ g \circ \rng{h}}
       &&\mbox{by RR.4,}\\
    &= \rng{h \circ \rng{h}}
       && \mbox{because $h \leq f \circ g$,}\\
    &= \rng{h} && \mbox{by RR.2.}
\end{align*}
\end{enumerate}
\end{proof}
