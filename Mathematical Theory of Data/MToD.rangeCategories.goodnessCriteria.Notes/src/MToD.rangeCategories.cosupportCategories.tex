\section{Cosupport Categories}

\begin{definition}
A \term{cosupport category} is a category $\catc$ equipped, for each
morphism $f:a\morph b$, with morphisms
\[
\rng{f}:b\morph b,
\]
called the \term{cosupport} of $f$.

The cosupport operator are the duals to the support axioms. We denote by $\mathrm{C}.1$--$\mathrm{C}.4$.
Thus, explicitly,

\noindent [C.1]\quad
If $f : a \morph b$, then
\[
f\circ\rng{f} = f.
\]

\noindent [C.2]\quad
If \fgsinkdiag in $\catc$, then
\[
\rng{f}\circ\rng{g}
=
\rng{g}\circ\rng{f}.
\]

\noindent [C.3]\quad
If \fgsinkdiag in $\catc$, then
\[
\rng{f \circ \rng{g}} = \rng{f}\circ\rng{g}.
\]

\noindent [C.4]\quad
If $\sequentialdiag{a}{b}{c}{f}{g}$ in $\catc$, then
\[
\rng{f\circ g} = \rng{f\circ \rng{g}}.
\]
\end{definition}

\begin{lemma}
\llabel{cosupportLemma}
If \catcw is a cosupport category  then 
\begin{enumerate}[(i)] 
\item For any object $a$ of \catc, 
\[
{\rng{id_a}= id_a},
\]
\item For any $f:a \morph b$ in in \catc, 
\[
\rng{\rng{f}}=\rng{f},
\]
\item For any $f:a \morph b$ in in \catc, 
\[
\rng{f} \comp \rng{f} = \rng{f},
\]
\item %old fglemma(ii) 
 \commentary {i.e. $\rng{f \circ g} \cleq \rng{g}$.}
 For $\sequentialdiag{a}{b}{c}{f}{g}$ in \catc,
\[
\rng{f \circ g} \circ \rng{g}=  \rng{f \circ g},
\]
\item %new
For \fgsinkdiag in \catc,
\[
f \circ \rng{g \circ \rng{f}} = f \circ \rng{g},
\]
\item %old fglemma (vii) 
 For  $\binarysinkContinued{a}{b}{c}{d}{f}{g}{h}$
in \catc,
\[
\rng{ f \circ \rng{g} \circ h } = \rng{ g \circ \rng{f} \circ h }.
\]
\end{enumerate}
\end{lemma}
\begin{proof}Follows by duality from lemma \lref{supportLemma}.
\end{proof}
